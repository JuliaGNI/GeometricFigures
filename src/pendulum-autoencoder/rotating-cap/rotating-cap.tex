% The exact encodings of the two rotating families,
% laid out as angle-atlas: M on top, in the same projection, the charts
% below it, and arrows from a point of each patch to its image.
%   phi_-: the chart r^2/2 = -p_theta at polar angle theta of {p_theta < 0}, which encodes family
%          (ii). The orbit at energy H is r(theta) = sqrt(2 sqrt(2(H - cos theta))) and encloses
%          exactly its action. Its decoder is the inverse of phi_- on r >= r_b, i.e. on
%          V_- = {p_theta < -1.05} (r_b = 1.4496, r_b^2/2 = 1.05, rho_1 = 1.45 in the cap), and
%          the cap of (ii) on the hole r < r_b.
%   phi_+: its mirror image under S(q1,q2,p1,p2) = (-q1,q2,-p1,p2), which sends iota(theta,p) to
%          iota(-theta,-p): r^2/2 = p_theta at polar angle -theta, V_+ = {p_theta > 1.05}, the
%          cap of (iii) on the hole.
% Both latent planes at true area, 1 unit = 0.55cm. Top left and right: the two decoders' images,
% drawn at the heights p_theta they have on M, each closed towards the strip |p_theta| < 1.05 by
% its own cap; the caps lie in R^4, not on M, and are drawn as domes over the boundary circles.
% Levels: H = 2 (the orbit R of angle-atlas, and its mirror image) and H = 4 on each family.
%
% Projection as in angle-atlas: u = theta - pi, turned by \cylA, a point is in front where
% u in [-120, 60]. On M, p_theta = +-sqrt(2(H + cos u)).
\newcommand{\cylR}{1.35}
\newcommand{\cylC}{0.24}
\newcommand{\cylS}{0.52}
\newcommand{\cylP}{3.3}     % where the picture cuts M, at p_theta = -3.3 and +3.3
\newcommand{\cylA}{30}
\newcommand{\pb}{1.05}      % the boundaries of V_-, V_+ at p_theta = -1.05 and +1.05
\newcommand{\rb}{1.4496}    % latent radius of the hole
\newcommand{\domeh}{0.95}      % height of a drawn dome
\pgfmathdeclarefunction{cylx}{1}{\pgfmathparse{\cylR*sin((#1)+\cylA)}}
\pgfmathdeclarefunction{cyly}{2}{\pgfmathparse{-\cylC*\cylR*cos((#1)+\cylA)+\cylS*(#2)}}
\pgfmathdeclarefunction{orb}{2}{\pgfmathparse{sqrt(2*((#2)+cos(#1)))}}            % |p| on M, angle u
\pgfmathdeclarefunction{rl}{2}{\pgfmathparse{sqrt(2*sqrt(2*((#2)-cos(#1))))}}     % latent radius, angle theta

% a band of M over u in [#2,#3], wholly in front or behind, for p_theta in (#4,#5)
\newcommand{\band}[5]{%
  \fill[#1]
    plot[domain=#2:#3,samples=40,smooth] ({cylx(\x)},{cyly(\x,#5)})
    -- plot[domain=#3:#2,samples=40,smooth] ({cylx(\x)},{cyly(\x,#4)}) -- cycle;}
\newcommand{\circleat}[2]{\draw[#1] plot[domain=-180:180,samples=120,smooth] ({cylx(\x)},{cyly(\x,#2)});}
% the orbits at H = 2 and 4 on the side #1 = -1 (family (ii)) or +1 (family (iii))
\newcommand{\orbits}[1]{%
  \foreach \H/\w in {2/very thick,4/thick}{
    \draw[rot,\w] plot[domain=-120:60,samples=70,smooth] ({cylx(\x)},{cyly(\x,#1*orb(\x,\H))});
    \draw[rot,\w,hidden] plot[domain=60:240,samples=70,smooth] ({cylx(\x)},{cyly(\x,#1*orb(\x,\H))});}}
% the cylinder's outline between the heights #1 < #2, and its cut at #3 (dotted)
\newcommand{\outline}[3]{%
  \draw[fg!25!bg] ({-\cylR},{\cylS*(#1)}) -- ({-\cylR},{\cylS*(#2)});
  \draw[fg!25!bg] ({\cylR},{\cylS*(#1)}) -- ({\cylR},{\cylS*(#2)});
  \circleat{fg!25!bg,densely dotted}{#3}}
% a cap: the disk over the circle at height #1, domed upwards (#2 = 1) or downwards (#2 = -1)
\newcommand{\capdome}[2]{%
  \fill[net!16!bg] plot[domain=-180:180,samples=120,smooth] ({cylx(\x)},{cyly(\x,#1)}) -- cycle;
  \fill[net!16!bg] ({-\cylR},{\cylS*(#1)}) arc[start angle=180,end angle={180-#2*180},x radius=\cylR,y radius=\domeh]
    -- cycle;
  \draw[net,thick] ({-\cylR},{\cylS*(#1)}) arc[start angle=180,end angle={180-#2*180},x radius=\cylR,y radius=\domeh];
  \draw[net,thick] plot[domain=-120:60,samples=70,smooth] ({cylx(\x)},{cyly(\x,#1)});
  \draw[net,thick,hidden] plot[domain=60:240,samples=70,smooth] ({cylx(\x)},{cyly(\x,#1)});}

% a latent plane: polar angle #1*theta, the image of V_-+ coloured #2 (to where the picture stops),
% the label of the green disk #3; the area the H = 2 orbit encloses, green disk included, is
% hatched in the left plane only (#1 = 1)
\newcommand{\latent}[3]{%
  \fill[#2!14!bg] (0,0) circle (2.75);
  \fill[net!16!bg] (0,0) circle (\rb);
  \ifnum#1=1
  \fill[pattern=north east lines,pattern color=rot!60!bg,smooth,samples=180,domain=0:360,variable=\t]
    plot ({rl(\t,2)*cos(#1*\t-30)},{rl(\t,2)*sin(#1*\t-30)}) -- cycle;
  \fi
  \draw[net,thick] (0,0) circle (\rb);
  \foreach \H/\w in {2/very thick,4/thick}{
    \draw[rot,\w,smooth,samples=180,domain=0:360,variable=\t]
      plot ({rl(\t,\H)*cos(#1*\t-30)},{rl(\t,\H)*sin(#1*\t-30)}) -- cycle;}
  \node[net!70!fg,lab,align=center,fill=net!16!bg,fill opacity=0.85,text opacity=1,inner sep=0.5pt] at (0,0) {#3};}
